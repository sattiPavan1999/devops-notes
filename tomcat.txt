Install tomcat on ec2 amazon linux




# Install Java
yum install java-17-amazon-corretto -y

# Install Git
yum install git -y

# Download Tomcat 9.0.117
wget https://dlcdn.apache.org/tomcat/tomcat-9/v9.0.117/bin/apache-tomcat-9.0.117.tar.gz

# Extract
tar -zxvf apache-tomcat-9.0.117.tar.gz

# Configure Tomcat Manager user
sed -i '56a\<role rolename="manager-gui"/>' apache-tomcat-9.0.117/conf/tomcat-users.xml
sed -i '57a\<role rolename="manager-script"/>' apache-tomcat-9.0.117/conf/tomcat-users.xml
sed -i '58a\<user username="tomcat" password="admin@123" roles="manager-gui,manager-script"/>' apache-tomcat-9.0.117/conf/tomcat-users.xml
sed -i '59a\</tomcat-users>' apache-tomcat-9.0.117/conf/tomcat-users.xml
sed -i '56d' apache-tomcat-9.0.117/conf/tomcat-users.xml

# Allow remote access to Manager App
sed -i '21d' apache-tomcat-9.0.117/webapps/manager/META-INF/context.xml
sed -i '22d' apache-tomcat-9.0.117/webapps/manager/META-INF/context.xml

# Start Tomcat
sh apache-tomcat-9.0.117/bin/startup.sh



for jenkins executors:

sudo mkdir -p /var/tmp_disk 
sudo chmod 1777 /var/tmp_disk 
sudo mount --bind /var/tmp_disk /tmp 
echo '/var/tmp_disk /tmp none bind 0 0' | sudo tee -a /etc/fstab 
sudo systemctl mask tmp.mount 
df -h /tmp 
sudo systemctl restart jenkins
